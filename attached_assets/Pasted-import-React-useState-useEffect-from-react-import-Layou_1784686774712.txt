                        className="bg-slate-900 border border-slate-700 rounded-xl px-4 py-2 text-sm text-white placeholder-slate-500 focus:outline-none focus:border-indigo-500"
                      />
                    </div>
                    <button
                      type="submit"
                      className="w-full py-2.5 bg-emerald-600 hover:bg-emerald-500 text-white font-semibold text-sm rounded-xl shadow-lg shadow-emerald-600/20 transition-all flex items-center justify-center gap-2"
                    >
                      <Plus className="w-4 h-4" /> Provision GHL + GNIL Instance
                    </button>
                  </form>
                </div>

              </div>

              {/* Tenants Management Table */}
              <div className="bg-slate-800/50 border border-slate-700/60 rounded-2xl overflow-hidden shadow-xl">
                <div className="p-6 border-b border-slate-700/60 flex justify-between items-center">
                  <div>
                    <h3 className="text-lg font-bold text-white">Multi-Tenant Subdomain Directory</h3>
                    <p className="text-xs text-slate-400 mt-0.5">Manage active client subdomains, licensing plans, and operational status.</p>
                  </div>
                  <span className="text-xs bg-indigo-500/10 border border-indigo-500/20 text-indigo-400 font-semibold px-3 py-1 rounded-full">
                    {tenants.length} Total Tenants
                  </span>
                </div>

                <div className="overflow-x-auto">
                  <table className="w-full text-left border-collapse">
                    <thead>
                      <tr className="border-b border-slate-700/60 bg-slate-900/60 text-slate-400 text-xs font-semibold uppercase tracking-wider">
                        <th className="py-4 px-6">Tenant Name</th>
                        <th className="py-4 px-6">Subdomain Endpoint</th>
                        <th className="py-4 px-6">Assigned Plan</th>
                        <th className="py-4 px-6">Monthly MRR</th>
                        <th className="py-4 px-6">Status</th>
                        <th className="py-4 px-6 text-right">Actions</th>
                      </tr>
                    </thead>
                    <tbody className="divide-y divide-slate-700/40 text-sm">
                      {tenants.map((t) => (
                        <tr key={t.id} className="hover:bg-slate-700/20 transition-colors">
                          <td className="py-4 px-6 font-semibold text-white flex items-center gap-3">
                            <div className="w-8 h-8 rounded-lg bg-indigo-600/20 border border-indigo-500/30 flex items-center justify-center text-indigo-400 font-bold text-xs">
                              {t.name.charAt(0)}
                            </div>
                            {t.name}
                          </td>
                          <td className="py-4 px-6 text-slate-400 font-mono text-xs">{t.subdomain}.getnextinline.io</td>
                          <td className="py-4 px-6 text-slate-300">{t.plan}</td>
                          <td className="py-4 px-6 font-semibold text-emerald-400">${t.mrr}/mo</td>
                          <td className="py-4 px-6">
                            <span className={`px-2.5 py-1 rounded-full text-xs font-semibold border ${t.status === 'Active' ? 'bg-emerald-500/10 border-emerald-500/20 text-emerald-400' : 'bg-rose-500/10 border-rose-500/20 text-rose-400'}`}>
                              {t.status}
                            </span>
                          </td>
                          <td className="py-4 px-6 text-right">
                            <button
                              onClick={() => toggleTenantStatus(t.id)}
                              className={`px-3 py-1.5 rounded-lg text-xs font-medium transition-all ${t.status === 'Active' ? 'bg-rose-500/10 hover:bg-rose-500/20 text-rose-400 border border-rose-500/30' : 'bg-emerald-500/10 hover:bg-emerald-500/20 text-emerald-400 border border-emerald-500/30'}`}
                            >
                              {t.status === 'Active' ? 'Suspend Tenant' : 'Activate Tenant'}
                            </button>
                          </td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              </div>

            </div>
          ) : (
            renderModuleConsole(activeTab)
          )}

        </main>
      </div>

      {/* Simulated Stripe Checkout Modal */}
      {stripeModal.show && (
        <div className="fixed inset-0 z-50 bg-slate-950/80 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="bg-slate-900 border border-slate-700 rounded-3xl max-w-md w-full p-6 space-y-6 shadow-2xl animate-in fade-in zoom-in-95 duration-200">
            <div className="flex justify-between items-center">
              <div className="flex items-center gap-3">
                <div className="p-2.5 bg-indigo-600/20 border border-indigo-500/30 rounded-xl text-indigo-400">
                  <CreditCard className="w-6 h-6" />
                </div>
                <div>
                  <h3 className="text-lg font-bold text-white">Stripe Checkout Simulation</h3>
                  <p className="text-xs text-slate-400">Secure Unified Gateway</p>
                </div>
              </div>
              <button
                onClick={() => setStripeModal({ show: false, moduleName: '', basePrice: 0, markupPrice: 0 })}
                className="text-slate-400 hover:text-white p-1 rounded-lg hover:bg-slate-800"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            <div className="p-4 bg-slate-800/60 rounded-2xl border border-slate-700/60 space-y-3">
              <div className="flex justify-between text-sm">
                <span className="text-slate-400">Selected Module/Service:</span>
                <span className="font-semibold text-white">{stripeModal.moduleName}</span>
              </div>
              <div className="flex justify-between text-sm">
                <span className="text-slate-400">Base Wholesale Cost:</span>
                <span className="font-mono text-slate-300">${stripeModal.basePrice}/mo</span>
              </div>
              <div className="flex justify-between text-sm">
                <span className="text-slate-400">Applied Agency Markup ({globalMarkup}%):</span>
                <span className="font-mono text-indigo-400">+${Math.round(stripeModal.basePrice * (globalMarkup / 100))}/mo</span>
              </div>
              <div className="pt-3 border-t border-slate-700 flex justify-between items-center">
                <span className="font-bold text-white">Total Resale Price:</span>
                <span className="text-xl font-black text-emerald-400">${stripeModal.markupPrice}/mo</span>
              </div>
            </div>

            <div className="space-y-3">
              <div className="p-3 bg-slate-800 border border-slate-700 rounded-xl flex items-center gap-3 text-xs text-slate-300">
                <CheckCircle className="w-4 h-4 text-emerald-400 flex-shrink-0" />
                <span>Test Mode Active: Simulating live tokenized bank transfer & subscription setup.</span>
              </div>
              <button
                onClick={completeStripePayment}
                className="w-full py-3 bg-indigo-600 hover:bg-indigo-500 text-white font-semibold rounded-xl shadow-lg shadow-indigo-600/30 transition-all flex items-center justify-center gap-2"
              >
                <Check className="w-5 h-5" /> Authorize & Provision (${stripeModal.markupPrice}/mo)
              </button>
            </div>
          </div>
        </div>
      )}

      {/* Toast Notification Banner */}
      {toast.show && (
        <div className="fixed bottom-6 right-6 z-50 bg-slate-900 border border-slate-700 text-white px-5 py-3 rounded-2xl shadow-2xl flex items-center gap-3 animate-in slide-in-from-bottom-5 duration-300">
          <CheckCircle className="w-5 h-5 text-emerald-400 flex-shrink-0" />
          <span className="text-sm font-medium">{toast.message}</span>
        </div>
      )}

    </div>
  );
}
